// 量两个角色的世界身高与尺度，用来把"武器相对身高的比例"对齐（而不是拿绝对长度硬套）
const string 杨戬Prefab = "Assets/resources/NPC/Human/YangJian/YangJian_03.prefab";

var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");
Transform 找(GameObject 根, string 名)
{
    foreach (var t in 根.GetComponentsInChildren<Transform>(true)) if (t.name == 名) return t;
    return null;
}
Bounds 全部渲染体(GameObject 根)
{
    bool 有 = false; var b = new Bounds();
    foreach (var r in 根.GetComponentsInChildren<Renderer>(true))
    {
        if (r is SkinnedMeshRenderer)
        {
            string n = r.name;
            if (n.Contains("PiaoDai") || n.Contains("FaHuan")) continue;   // 飘带/法环会虚高，不算身高
        }
        if (!有) { b = r.bounds; 有 = true; } else b.Encapsulate(r.bounds);
    }
    return b;
}

var 杨戬 = (GameObject)UnityEditor.PrefabUtility.InstantiatePrefab(UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>(杨戬Prefab));
杨戬.transform.position = new Vector3(0f, -500f, 0f);
var 杨戬体 = 全部渲染体(杨戬);
var 杨戬髋 = 找(杨戬, "Bip01 Pelvis"); var 杨戬头 = 找(杨戬, "Bip01 Head");
L("杨戬：根 lossyScale=" + 杨戬.transform.lossyScale.ToString("F4")
  + " Bip01 lossyScale=" + (找(杨戬, "Bip01") == null ? "-" : 找(杨戬, "Bip01").lossyScale.ToString("F4")));
L("杨戬：渲染体总包围盒 尺寸=" + 杨戬体.size.ToString("F3") + " 中心=" + 杨戬体.center.ToString("F3"));
L("杨戬：髋=" + 杨戬髋.position.ToString("F3") + " 头=" + 杨戬头.position.ToString("F3")
  + " 髋→头=" + Vector3.Distance(杨戬髋.position, 杨戬头.position).ToString("F3") + " 米");
var 杨戬武器 = 找(杨戬, "YangJian_03_Weapon").GetComponent<SkinnedMeshRenderer>();
var 杨戬手 = 找(杨戬, "Bip01 R Hand");
L("杨戬：武器最长=" + Mathf.Max(杨戬武器.bounds.size.x, Mathf.Max(杨戬武器.bounds.size.y, 杨戬武器.bounds.size.z)).ToString("F3")
  + " 武器中心离手=" + Vector3.Distance(杨戬武器.bounds.center, 杨戬手.position).ToString("F3"));
UnityEngine.Object.DestroyImmediate(杨戬);

var 真玩家 = GameObject.Find("Player");
var 副本 = UnityEngine.Object.Instantiate(真玩家);
副本.transform.position = new Vector3(0f, -500f, 0f);
foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
var an = 副本.GetComponentInChildren<Animator>(true);
an.runtimeAnimatorController = null;
var 玩家体 = 全部渲染体(副本);
var 玩家髋 = 找(副本, "Hip"); var 玩家头 = 找(副本, "Head");
L("");
L("玩家：渲染体总包围盒 尺寸=" + 玩家体.size.ToString("F3") + " 中心=" + 玩家体.center.ToString("F3"));
L("玩家：Hip=" + 玩家髋.position.ToString("F3") + " Head=" + 玩家头.position.ToString("F3")
  + " 髋→头=" + Vector3.Distance(玩家髋.position, 玩家头.position).ToString("F3") + " 米");
UnityEngine.Object.DestroyImmediate(副本);

L("");
L("→ 若两人的『渲染体高度』分别为上面两个 size.y，则武器应满足 『武器长 ÷ 身高』一致");
return sb.ToString();
