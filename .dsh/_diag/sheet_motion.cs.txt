// 诊断编排图：同一段动作取 4 个时间点、把 4 个副本并排摆好，一张图看完整条动作弧
// 目的：判断"重定向出来的动作很怪异"到底怪在哪（武器握持 / 腿 / 髋 / 整体姿态）
var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");
Transform 找(GameObject 根, string 名)
{
    foreach (var t in 根.GetComponentsInChildren<Transform>(true)) if (t.name == 名) return t;
    return null;
}
string 截图目录 = System.IO.Path.Combine(System.IO.Path.GetDirectoryName(Application.dataPath), "screenshots");

System.Action<Vector3, string> 拍一排 = (相机位, 文件名) =>
{
    var camGo = new GameObject("临时相机");
    var cam = camGo.AddComponent<Camera>();
    cam.fieldOfView = 34f; cam.nearClipPlane = 0.05f; cam.farClipPlane = 300f;
    cam.clearFlags = CameraClearFlags.SolidColor;
    cam.backgroundColor = new Color(0.12f, 0.13f, 0.15f, 1f);
    camGo.transform.position = 相机位;
    camGo.transform.LookAt(new Vector3(0f, -498.2f, 0f));
    var 灯Go = new GameObject("临时灯");
    var 灯 = 灯Go.AddComponent<Light>();
    灯.type = LightType.Directional; 灯.intensity = 1.5f;
    灯Go.transform.rotation = Quaternion.Euler(40f, 25f, 0f);
    var rt = new RenderTexture(1500, 620, 24);
    cam.targetTexture = rt; cam.Render();
    RenderTexture.active = rt;
    var tex = new Texture2D(1500, 620, TextureFormat.RGB24, false);
    tex.ReadPixels(new Rect(0f, 0f, 1500f, 620f), 0, 0); tex.Apply();
    RenderTexture.active = null; cam.targetTexture = null;
    System.IO.File.WriteAllBytes(System.IO.Path.Combine(截图目录, 文件名), tex.EncodeToPNG());
    UnityEngine.Object.DestroyImmediate(rt); UnityEngine.Object.DestroyImmediate(tex);
    UnityEngine.Object.DestroyImmediate(camGo); UnityEngine.Object.DestroyImmediate(灯Go);
};

var 真玩家 = GameObject.Find("Player");
var 武器资产 = UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>("Assets/resources/Weapons/八九玄功_三尖两刃刀.prefab");
string[] 片段 = new string[] { "技能动作/八九玄功/长刀女_Attack1", "技能动作/八九玄功/持刀_站", "技能动作/八九玄功/持刀_走", "技能动作/八九玄功/持刀_御风Idle" };
float[] 比例 = new float[] { 0.15f, 0.38f, 0.60f, 0.82f };

L("编排图：每张 = 同一动作 4 个时间点（15% / 38% / 60% / 82%）并排，正面看");

foreach (var 名 in 片段)
{
    var clip = Resources.Load<AnimationClip>(名);
    if (clip == null) { L("❌ 取不到 " + 名); continue; }

    var 副本们 = new System.Collections.Generic.List<GameObject>();
    for (int i = 0; i < 4; i++)
    {
        var 副本 = UnityEngine.Object.Instantiate(真玩家);
        副本.name = "编排_" + i;
        副本.transform.position = new Vector3(-5.4f + i * 3.6f, -500f, 0f);
        副本.transform.rotation = Quaternion.identity;
        foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
        var an = 副本.GetComponentInChildren<Animator>(true);
        an.runtimeAnimatorController = null;
        foreach (var s in 副本.GetComponentsInChildren<SkinnedMeshRenderer>(true)) s.forceMatrixRecalculationPerRender = true;

        var 手 = 找(副本, "R_Hand");
        var 武器 = UnityEngine.Object.Instantiate(武器资产);
        武器.transform.SetParent(手, false);
        武器.transform.localPosition = Vector3.zero;
        武器.transform.localRotation = Quaternion.identity;
        武器.transform.localScale = Vector3.one * 0.4911f;
        武器.GetComponentInChildren<SkinnedMeshRenderer>().forceMatrixRecalculationPerRender = true;

        clip.SampleAnimation(an.gameObject, clip.length * 比例[i]);
        副本们.Add(副本);
    }
    // 相机在 +Z 侧（角色正面），把 4 个人都框进去
    拍一排(new Vector3(0f, -497.9f, 13.2f), "编排_" + 名.Replace("技能动作/八九玄功/", "") + ".png");
    L("  ✅ " + 名 + "（" + clip.length.ToString("F2") + "s）→ 编排_" + 名.Replace("技能动作/八九玄功/", "") + ".png");
    foreach (var c in 副本们) UnityEngine.Object.DestroyImmediate(c);
}
return sb.ToString();
