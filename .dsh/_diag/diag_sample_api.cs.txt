// 对照两种采样 API：AnimationClip.SampleAnimation vs AnimationMode.SampleAnimationClip
// 判据：采样后 R_Upperarm 的 localRotation 是否等于曲线在该时刻的值
var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");
var clip = Resources.Load<AnimationClip>("技能动作/八九玄功/长刀女_Attack1");
if (clip == null) { L("❌ 取不到"); return sb.ToString(); }

// 曲线参照值
EditorCurveBinding 臂 = default(EditorCurveBinding); bool 有 = false;
foreach (var b in UnityEditor.AnimationUtility.GetCurveBindings(clip))
    if (b.path.EndsWith("R_Upperarm") && b.propertyName == "localRotation.x") { 臂 = b; 有 = true; break; }
var 臂曲 = 有 ? UnityEditor.AnimationUtility.GetEditorCurve(clip, 臂) : null;
L("R_Upperarm.localRotation.x 曲线：" + (臂曲 == null ? "❌ 没有" : "t0=" + 臂曲.Evaluate(0f).ToString("F4") + " t50%=" + 臂曲.Evaluate(clip.length * 0.5f).ToString("F4") + " 键=" + 臂曲.keys.Length));

var 副本 = UnityEngine.Object.Instantiate(GameObject.Find("Player"));
副本.name = "采样对照2";
副本.transform.position = new Vector3(0f, -600f, 0f);
foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
var an = 副本.GetComponentInChildren<Animator>(true);
an.runtimeAnimatorController = null;
foreach (var s in 副本.GetComponentsInChildren<SkinnedMeshRenderer>(true)) s.forceMatrixRecalculationPerRender = true;
Transform 臂骨 = null, 手 = null;
foreach (var t in 副本.GetComponentsInChildren<Transform>(true)) { if (t.name == "R_Upperarm") 臂骨 = t; if (t.name == "R_Hand") 手 = t; }
L("臂骨路径 = " + 臂骨.name + "（父 " + 臂骨.parent.name + "）");

L("");
L("A) clip.SampleAnimation：");
foreach (var 比例 in new float[] { 0f, 0.5f })
{
    clip.SampleAnimation(an.gameObject, clip.length * 比例);
    L("   " + (比例 * 100f).ToString("F0") + "%：臂骨 localRotation.x=" + 臂骨.localRotation.x.ToString("F4")
      + " 手世界=" + 手.position.ToString("F4"));
}

L("");
L("B) AnimationMode.SampleAnimationClip：");
UnityEditor.AnimationMode.StartAnimationMode();
try
{
    foreach (var 比例 in new float[] { 0f, 0.5f })
    {
        UnityEditor.AnimationMode.SampleAnimationClip(an.gameObject, clip, clip.length * 比例);
        L("   " + (比例 * 100f).ToString("F0") + "%：臂骨 localRotation.x=" + 臂骨.localRotation.x.ToString("F4")
          + " 手世界=" + 手.position.ToString("F4"));
    }
}
finally { UnityEditor.AnimationMode.StopAnimationMode(); }

UnityEngine.Object.DestroyImmediate(副本);
return sb.ToString();
