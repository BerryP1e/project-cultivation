// 用"身体网格（不含武器/飘带/法环）的世界高度"来定武器缩放比例
// 目标：武器长度 ÷ 身体高度 在两个角色上一致
const string 杨戬Prefab = "Assets/resources/NPC/Human/YangJian/YangJian_03.prefab";

var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");

var 杨戬 = (GameObject)UnityEditor.PrefabUtility.InstantiatePrefab(UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>(杨戬Prefab));
杨戬.transform.position = new Vector3(0f, -500f, 0f);
杨戬.transform.rotation = Quaternion.identity;
SkinnedMeshRenderer 杨戬身 = null, 杨戬武器 = null;
foreach (var s in 杨戬.GetComponentsInChildren<SkinnedMeshRenderer>(true))
{
    if (s.name == "YangJian_03") 杨戬身 = s;
    if (s.name == "YangJian_03_Weapon") 杨戬武器 = s;
}
L("杨戬 身体网格=" + (杨戬身 == null ? "没找到" : 杨戬身.name) + " 世界尺寸=" + (杨戬身 == null ? "-" : 杨戬身.bounds.size.ToString("F3")));
L("杨戬 武器网格 本地尺寸=" + 杨戬武器.localBounds.size.ToString("F4") + " 渲染体 lossyScale=" + 杨戬武器.transform.lossyScale.ToString("F4")
  + " → 武器世界长 ≈ " + (杨戬武器.localBounds.size.x * 杨戬武器.transform.lossyScale.x).ToString("F3") + " 米");
float 杨戬身高 = 杨戬身 == null ? 0f : 杨戬身.bounds.size.y;
float 杨戬武器长 = 杨戬武器.localBounds.size.x * 杨戬武器.transform.lossyScale.x;
L("比例（杨戬）：武器长 ÷ 身高 = " + 杨戬武器长.ToString("F3") + " / " + 杨戬身高.ToString("F3") + " = " + (杨戬武器长 / 杨戬身高).ToString("F3"));
UnityEngine.Object.DestroyImmediate(杨戬);

var 真玩家 = GameObject.Find("Player");
var 副本 = UnityEngine.Object.Instantiate(真玩家);
副本.transform.position = new Vector3(0f, -500f, 0f);
副本.transform.rotation = Quaternion.identity;
foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
var an = 副本.GetComponentInChildren<Animator>(true);
an.runtimeAnimatorController = null;
SkinnedMeshRenderer 玩家身 = null;
foreach (var s in 副本.GetComponentsInChildren<SkinnedMeshRenderer>(true)) { 玩家身 = s; break; }
L("");
L("玩家 身体网格=" + (玩家身 == null ? "没找到" : 玩家身.name) + " 世界尺寸=" + (玩家身 == null ? "-" : 玩家身.bounds.size.ToString("F3")));
float 玩家身高 = 玩家身 == null ? 0f : 玩家身.bounds.size.y;

L("");
L("=== 结论 ===");
L("杨戬：身高=" + 杨戬身高.ToString("F3") + " 武器长=" + 杨戬武器长.ToString("F3") + " 比例=" + (杨戬武器长 / 杨戬身高).ToString("F3"));
L("玩家：身高=" + 玩家身高.ToString("F3"));
L("玩家武器网格 本地长轴=4.9154（两个模型同一份网格）");
L("玩家武器世界长 = 4.9154 × 玩家手骨lossyScale(1.8007) × 缩放");
float 应有长 = 玩家身高 * (杨戬武器长 / 杨戬身高);
float 应缩放 = 应有长 / (4.9154f * 1.8007f);
L("要让『武器长 ÷ 身高』和杨戬一致：武器应有长 = " + 应有长.ToString("F3") + " 米 → 缩放 = " + 应缩放.ToString("F4"));
L("（第一版按绝对长度硬套得到的是 0.3846，差 " + ((0.3846f / 应缩放 - 1f) * 100f).ToString("F1") + "%）");
return sb.ToString();
