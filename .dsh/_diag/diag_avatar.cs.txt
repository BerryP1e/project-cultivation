// 把玩家 Avatar 的人形映射（HumanBodyBones → 骨骼名/路径）打出来，确认"脊椎"到底映到了哪根骨
var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");
var 副本 = UnityEngine.Object.Instantiate(GameObject.Find("Player"));
副本.transform.position = new Vector3(0f, -500f, 0f);
foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
var an = 副本.GetComponentInChildren<Animator>(true);
an.runtimeAnimatorController = null;
var av = an.avatar;
L("avatar=" + av.name + " isHuman=" + av.isHuman + " isValid=" + av.isValid);
var hd = av.humanDescription;
L("human 映射条数 = " + hd.human.Length + "，skeleton 条数 = " + hd.skeleton.Length);
L("");
L("=== 关心的几根 ===");
string[] 关心 = new string[] { "Hips", "Spine", "Chest", "UpperChest", "Neck", "Head", "LeftUpperLeg", "RightUpperLeg", "LeftFoot", "LeftShoulder", "RightHand", "LeftHand" };
foreach (var h in hd.human)
{
    string n = h.boneName;
    foreach (var k in 关心)
        if (string.Equals(h.humanName.Replace(" ", ""), k, System.StringComparison.OrdinalIgnoreCase))
            L("   " + h.humanName.PadRight(16) + " ← " + n);
}
L("");
L("=== 全部映射（humanName ← boneName）===");
foreach (var h in hd.human) L("   " + h.humanName.PadRight(18) + " ← " + h.boneName);
UnityEngine.Object.DestroyImmediate(副本);
return sb.ToString();
