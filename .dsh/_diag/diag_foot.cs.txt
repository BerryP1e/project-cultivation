// 定位"烘焙片段 vs 纯御风 Idle"的脚部偏差来自哪里：根变换？髋？还是肌肉？
const string 御风Idle路径 = "Assets/Animations/凭虚御风/御风_Idle.anim";
const string 烘焙 = "Assets/Animations/八九玄功/长刀女_Attack1_御风.anim";
const string 攻击 = "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack1.FBX";

var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");
Transform 找(GameObject 根, string 名)
{
    foreach (var t in 根.GetComponentsInChildren<Transform>(true)) if (t.name == 名) return t;
    return null;
}
System.Collections.Generic.Dictionary<string, AnimationCurve> 读曲线(AnimationClip c)
{
    var d = new System.Collections.Generic.Dictionary<string, AnimationCurve>();
    foreach (var b in UnityEditor.AnimationUtility.GetCurveBindings(c))
        if (b.path == "") { var cur = UnityEditor.AnimationUtility.GetEditorCurve(c, b); if (cur != null) d[b.propertyName] = cur; }
    return d;
}

var 副本 = UnityEngine.Object.Instantiate(GameObject.Find("Player"));
副本.transform.position = new Vector3(0f, -500f, 0f);
副本.transform.rotation = Quaternion.identity;
foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
var an = 副本.GetComponentInChildren<Animator>(true);
an.runtimeAnimatorController = null;
foreach (var s in 副本.GetComponentsInChildren<SkinnedMeshRenderer>(true)) s.forceMatrixRecalculationPerRender = true;

var 新 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(烘焙);
var 御 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(御风Idle路径);
var 攻 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(攻击);
var 新曲 = 读曲线(新); var 御曲 = 读曲线(御);
L("曲线数：新=" + 新曲.Count + " 御风=" + 御曲.Count + " 攻=" + 读曲线(攻).Count);
var 只在新 = new System.Collections.Generic.List<string>();
foreach (var k in 新曲.Keys) if (!御曲.ContainsKey(k)) 只在新.Add(k);
var 只在御 = new System.Collections.Generic.List<string>();
foreach (var k in 御曲.Keys) if (!新曲.ContainsKey(k)) 只在御.Add(k);
L("只在新片段里有：" + (只在新.Count == 0 ? "无" : string.Join(",", 只在新.ToArray())));
L("只在御风里有：" + (只在御.Count == 0 ? "无" : string.Join(",", 只在御.ToArray())));

var 髋 = 找(副本, "Hip"); var 盆 = 找(副本, "Pelvis"); var 踝 = 找(副本, "L_Foot"); var 手 = 找(副本, "R_Hand");
var 视觉 = 找(副本, "Player_Visual");
L("");
foreach (var 比例 in new float[] { 0.5f, 0.7f })
{
    float t = 新.length * 比例;
    L("=== t=" + t.ToString("F3") + "s（御风采样点 " + Mathf.Repeat(t, 御.length).ToString("F3") + "s）");
    新.SampleAnimation(an.gameObject, t);
    L("  新  ：视觉根 localPos=" + 视觉.localPosition.ToString("F4") + " | Hip=" + 髋.localRotation.eulerAngles.ToString("F1")
      + " 盆=" + 盆.localRotation.eulerAngles.ToString("F1") + " 踝=" + 踝.position.ToString("F4") + " 手=" + 手.position.ToString("F4")
      + " | 大腿FB肌肉曲线=" + (新曲.ContainsKey("Left Upper Leg Front-Back") ? 新曲["Left Upper Leg Front-Back"].Evaluate(t).ToString("F5") : "-"));
    御.SampleAnimation(an.gameObject, Mathf.Repeat(t, 御.length));
    L("  御风：视觉根 localPos=" + 视觉.localPosition.ToString("F4") + " | Hip=" + 髋.localRotation.eulerAngles.ToString("F1")
      + " 盆=" + 盆.localRotation.eulerAngles.ToString("F1") + " 踝=" + 踝.position.ToString("F4") + " 手=" + 手.position.ToString("F4")
      + " | 大腿FB肌肉曲线=" + (御曲.ContainsKey("Left Upper Leg Front-Back") ? 御曲["Left Upper Leg Front-Back"].Evaluate(Mathf.Repeat(t, 御.length)).ToString("F5") : "-"));
    L("  → 踝差=" + Vector3.Distance(踝.position, 踝.position).ToString("F3") + "（打印顺序所限，见上面两行逐项对比）");
}
UnityEngine.Object.DestroyImmediate(副本);
return sb.ToString();
