var player = GameObject.Find("Player");
var anim = player.GetComponentInChildren<Animator>();
var driver = player.GetComponent<PlayerAnimationController>();
var pd = GameObject.Find("CharacterUI").GetComponent<UIPanelData>();
var gf = UnityEditor.AssetDatabase.LoadAssetAtPath<GongFaDefinition>("Assets/Data/Generated/GongFaDefinition/gongfa_jiuba_xuangong.asset");
var lines = new System.Text.StringBuilder();
System.Collections.IEnumerator Run()
{
 pd.学会(gf); pd.当前功法=gf; pd.RaiseChanged();
 yield return new WaitForSecondsRealtime(1);
 foreach(var name in new[]{"长刀女_Attack1","长刀女_Attack2","杨戬_Attack1","杨戬_Attack2"})
 {
  var clip=Resources.Load<AnimationClip>("技能动作/八九玄功/"+name);
  driver.PlayAction(clip);
  float end=Time.time+clip.length+0.6f, min=999,max=-999, step=0,last=anim.transform.position.y, foot=999,rootMin=999,rootMax=-999;
  while(Time.time<end)
  {
   yield return new WaitForEndOfFrame();
   float y=anim.GetBoneTransform(HumanBodyBones.Hips).position.y-player.transform.position.y;
   min=Mathf.Min(min,y);max=Mathf.Max(max,y);
   float root=anim.transform.position.y-player.transform.position.y;
   rootMin=Mathf.Min(rootMin,root);rootMax=Mathf.Max(rootMax,root);
   step=Mathf.Max(step,Mathf.Abs(anim.transform.position.y-last));last=anim.transform.position.y;
   foot=Mathf.Min(foot,Mathf.Min(anim.GetBoneTransform(HumanBodyBones.LeftFoot).position.y,anim.GetBoneTransform(HumanBodyBones.RightFoot).position.y)-player.transform.position.y);
  }
  lines.AppendLine(name+" hips="+min.ToString("F3")+".."+max.ToString("F3")+" visual="+rootMin.ToString("F3")+".."+rootMax.ToString("F3")+" maxFrameStep="+step.ToString("F3")+" footMin="+foot.ToString("F3"));
 }
 System.IO.File.WriteAllText("screenshots/jiuba-motion-audit.txt",lines.ToString());
 Debug.Log(lines.ToString());
}
return Run();
