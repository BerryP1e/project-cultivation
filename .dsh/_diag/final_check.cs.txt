// 收尾总检：武器 prefab + 四个源片段 + 四个烘焙片段，逐项打表
const string 武器Prefab路径 = "Assets/resources/Weapons/八九玄功_三尖两刃刀.prefab";
const string 烘焙目录 = "Assets/Animations/八九玄功";
var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");

var 武器 = UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>(武器Prefab路径);
if (武器 == null) L("❌ 武器 prefab 加载失败");
else
{
    var smr = 武器.GetComponentInChildren<SkinnedMeshRenderer>();
    var 骨 = 武器.transform.Find("Bone01");
    L("武器 prefab ✅ " + 武器Prefab路径);
    L("   子节点=" + 武器.GetComponentsInChildren<Transform>(true).Length
      + " 骨骼=" + (骨 == null ? "缺" : 骨.name)
      + " 网格=" + (smr == null ? "缺" : smr.sharedMesh.name)
      + " 材质=" + (smr == null || smr.sharedMaterial == null ? "缺" : smr.sharedMaterial.name)
      + " 顶点=" + (smr == null || smr.sharedMesh == null ? 0 : smr.sharedMesh.vertexCount)
      + " 蒙皮长轴=" + (smr == null ? 0f : smr.localBounds.size.x).ToString("F3")
      + " rootBone=" + (smr == null || smr.rootBone == null ? "缺" : smr.rootBone.name)
      + " bones=" + (smr == null ? 0 : smr.bones.Length));
}

string[] 源 = new string[] {
    "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack1.FBX",
    "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack2.FBX",
    "Assets/resources/NPC/Human/YangJian_Animation/YangJian@Attack1.FBX",
    "Assets/resources/NPC/Human/YangJian_Animation/YangJian@Attack2.FBX" };
string[] 烘焙名 = new string[] { "长刀女_Attack1_御风", "长刀女_Attack2_御风", "杨戬_Attack1_御风", "杨戬_Attack2_御风" };

L("");
L("源片段 → 烘焙片段");
for (int i = 0; i < 源.Length; i++)
{
    var a = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(源[i]);
    var b = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(烘焙目录 + "/" + 烘焙名[i] + ".anim");
    L("  " + System.IO.Path.GetFileName(源[i]).Replace(".FBX", "") + "：" + (a == null ? "❌" : a.length.ToString("F2") + "s humanMotion=" + a.humanMotion + " 曲线=" + UnityEditor.AnimationUtility.GetCurveBindings(a).Length)
      + "  →  " + 烘焙名[i] + "：" + (b == null ? "❌" : b.length.ToString("F2") + "s humanMotion=" + b.humanMotion + " 曲线=" + UnityEditor.AnimationUtility.GetCurveBindings(b).Length
        + " loop=" + UnityEditor.AnimationUtility.GetAnimationClipSettings(b).loopTime));
}
return sb.ToString();
