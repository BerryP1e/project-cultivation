// 验证（第二版）：把武器挂到玩家右手，采样四段攻击动作，量尺寸/握持并拍照
// ⚠️ 全部在【临时副本】上做，场景里的 Player 一个字节都不动
// 修正一：SampleAnimation 必须打给【挂 Animator 的那个 GameObject】（Player_Visual），打给副本根无效（实测姿势不变）
// 修正二：武器缩放按实测比例算（杨戬本体手上的长度 ÷ 玩家手上的长度）
const string 武器Prefab路径 = "Assets/resources/Weapons/八九玄功_三尖两刃刀.prefab";
const string 杨戬Prefab = "Assets/resources/NPC/Human/YangJian/YangJian_03.prefab";
string[] 片段路径 = new string[] {
    "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack1.FBX",
    "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack2.FBX",
    "Assets/resources/NPC/Human/YangJian_Animation/YangJian@Attack1.FBX",
    "Assets/resources/NPC/Human/YangJian_Animation/YangJian@Attack2.FBX" };
float[] 采样点 = new float[] { 0.30f, 0.45f, 0.60f, 0.80f };

var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");

Transform 找(GameObject 根, string 名)
{
    foreach (var t in 根.GetComponentsInChildren<Transform>(true)) if (t.name == 名) return t;
    return null;
}
float 最长(Bounds b) => Mathf.Max(b.size.x, Mathf.Max(b.size.y, b.size.z));

System.Action<GameObject, Vector3, string> 拍 = (目标, 偏移, 路径) =>
{
    var camGo = new GameObject("临时相机");
    var cam = camGo.AddComponent<Camera>();
    cam.fieldOfView = 38f; cam.nearClipPlane = 0.05f; cam.farClipPlane = 200f;
    cam.clearFlags = CameraClearFlags.SolidColor;
    cam.backgroundColor = new Color(0.13f, 0.14f, 0.16f, 1f);
    camGo.transform.position = 目标.transform.position + 偏移;
    camGo.transform.LookAt(目标.transform.position + new Vector3(0f, 1.05f, 0f));
    var 灯Go = new GameObject("临时灯");
    var 灯 = 灯Go.AddComponent<Light>();
    灯.type = LightType.Directional; 灯.intensity = 1.35f; 灯.color = Color.white;
    灯Go.transform.rotation = Quaternion.Euler(48f, 35f, 0f);
    var rt = new RenderTexture(760, 1010, 24);
    cam.targetTexture = rt;
    cam.Render();
    RenderTexture.active = rt;
    var tex = new Texture2D(760, 1010, TextureFormat.RGB24, false);
    tex.ReadPixels(new Rect(0f, 0f, 760f, 1010f), 0, 0);
    tex.Apply();
    RenderTexture.active = null; cam.targetTexture = null;
    System.IO.File.WriteAllBytes(路径, tex.EncodeToPNG());
    UnityEngine.Object.DestroyImmediate(rt);
    UnityEngine.Object.DestroyImmediate(tex);
    UnityEngine.Object.DestroyImmediate(camGo);
    UnityEngine.Object.DestroyImmediate(灯Go);
};

// ---- 参照：武器在杨戬自己手上多长 ----
var 杨戬 = (GameObject)UnityEditor.PrefabUtility.InstantiatePrefab(UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>(杨戬Prefab));
杨戬.transform.position = new Vector3(0f, -500f, 0f);
杨戬.transform.rotation = Quaternion.identity;
var 杨戬武器Smr = 找(杨戬, "YangJian_03_Weapon").GetComponent<SkinnedMeshRenderer>();
float 参照长 = 最长(杨戬武器Smr.bounds);
L("参照（杨戬本体，缩放1静止）：武器世界尺寸=" + 杨戬武器Smr.bounds.size.ToString("F3") + " → 最长 " + 参照长.ToString("F3") + " 米");
UnityEngine.Object.DestroyImmediate(杨戬);

// ---- 临时玩家副本 ----
var 真玩家 = GameObject.Find("Player");
if (真玩家 == null) { L("场景里没有 Player"); return sb.ToString(); }
var 副本 = UnityEngine.Object.Instantiate(真玩家);
副本.name = "临时验证_玩家";
副本.transform.position = new Vector3(0f, -500f, 0f);
副本.transform.rotation = Quaternion.identity;
foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
var an = 副本.GetComponentInChildren<Animator>(true);
an.runtimeAnimatorController = null;      // 清控制器，别每帧抢姿势；Animator 保持 enabled（人形采样靠它的 avatar）
var 采样目标 = an.gameObject;             // ★ 必须打给挂 Animator 的那个对象
L("采样目标 = " + 采样目标.name + "（avatar=" + (an.avatar == null ? "null" : an.avatar.name) + "）");

var 手 = 找(副本, "R_Hand");
var 武器 = UnityEngine.Object.Instantiate(UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>(武器Prefab路径));
武器.name = "八九玄功_三尖两刃刀";
武器.transform.SetParent(手, false);
武器.transform.localPosition = Vector3.zero;
武器.transform.localRotation = Quaternion.identity;
武器.transform.localScale = Vector3.one;
var 武器Smr = 武器.GetComponentInChildren<SkinnedMeshRenderer>();

float 当前长 = 最长(武器Smr.bounds);
float 缩放 = 参照长 / 当前长;
L("玩家挂载后（缩放1）：武器世界尺寸=" + 武器Smr.bounds.size.ToString("F3") + " → 最长 " + 当前长.ToString("F3") + " 米");
L("按参照算出的缩放 = " + 参照长.ToString("F3") + " / " + 当前长.ToString("F3") + " = " + 缩放.ToString("F4"));
武器.transform.localScale = Vector3.one * 缩放;
武器Smr = 武器.GetComponentInChildren<SkinnedMeshRenderer>();
L("缩放后：武器世界尺寸=" + 武器Smr.bounds.size.ToString("F3") + " → 最长 " + 最长(武器Smr.bounds).ToString("F3") + " 米"
  + "，中心离手=" + Vector3.Distance(武器Smr.bounds.center, 手.position).ToString("F3") + " 米");

string 截图目录 = System.IO.Path.Combine(System.IO.Path.GetDirectoryName(Application.dataPath), "screenshots");
System.IO.Directory.CreateDirectory(截图目录);

// 先验一次采样到底有没有生效
var 测试片段 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(片段路径[2]);
测试片段.SampleAnimation(采样目标, 0f);
var 手0 = 手.position;
测试片段.SampleAnimation(采样目标, 测试片段.length * 0.55f);
L("采样自检：" + 测试片段.name + " t=0 手=" + 手0.ToString("F3") + " → t=55% 手=" + 手.position.ToString("F3")
  + " 位移=" + Vector3.Distance(手0, 手.position).ToString("F3") + " 米 " + (Vector3.Distance(手0, 手.position) > 0.02f ? "✅ 采样生效" : "❌ 采样仍然没生效"));

foreach (var p in 片段路径)
{
    var clip = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(p);
    if (clip == null) { L("拿不到片段 " + p); continue; }
    string 短名 = System.IO.Path.GetFileNameWithoutExtension(p).Replace("@", "_");
    L("");
    L("--- " + clip.name + "（" + clip.length.ToString("F2") + "s）");
    foreach (var 比例 in 采样点)
    {
        float t = clip.length * 比例;
        clip.SampleAnimation(采样目标, t);
        武器Smr = 武器.GetComponentInChildren<SkinnedMeshRenderer>();
        L("   " + (比例 * 100f).ToString("F0") + "%（t=" + t.ToString("F2") + "s）：武器最长=" + 最长(武器Smr.bounds).ToString("F2")
          + " 中心离手=" + Vector3.Distance(武器Smr.bounds.center, 手.position).ToString("F2")
          + " 手世界=" + 手.position.ToString("F3") + " 武器中心=" + 武器Smr.bounds.center.ToString("F3"));
    }
    clip.SampleAnimation(采样目标, clip.length * 0.55f);
    拍(副本, new Vector3(3.3f, 1.75f, -3.4f), System.IO.Path.Combine(截图目录, "八九_" + 短名 + "_斜前.png"));
    拍(副本, new Vector3(4.3f, 0.55f, 0f), System.IO.Path.Combine(截图目录, "八九_" + 短名 + "_正侧.png"));
}

UnityEngine.Object.DestroyImmediate(武器);
UnityEngine.Object.DestroyImmediate(副本);
L("");
L("临时对象已销毁（场景里的 Player 未改动、未保存）");
return sb.ToString();
