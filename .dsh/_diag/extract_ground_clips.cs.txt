// 把四个「地面版」攻击片段从 FBX 里抽成独立 .anim，放进 resources/技能动作/八九玄功/
// 理由：普攻组件统一按 Resources.Load<AnimationClip>("技能动作/…") 取片段（和 _御风 版一致）
var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");

const string 目标目录 = "Assets/resources/技能动作/八九玄功";
string[,] 表 = new string[,] {
    { "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack1.FBX", "长刀女_Attack1" },
    { "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack2.FBX", "长刀女_Attack2" },
    { "Assets/resources/NPC/Human/YangJian_Animation/YangJian@Attack1.FBX", "杨戬_Attack1" },
    { "Assets/resources/NPC/Human/YangJian_Animation/YangJian@Attack2.FBX", "杨戬_Attack2" },
};

if (!UnityEditor.AssetDatabase.IsValidFolder(目标目录))
    L("建目录 → " + UnityEditor.AssetDatabase.CreateFolder("Assets/resources/技能动作", "八九玄功"));

for (int i = 0; i < 表.GetLength(0); i++)
{
    string 源 = 表[i, 0];
    string 出 = 目标目录 + "/" + 表[i, 1] + ".anim";
    var 原 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(源);
    if (原 == null) { L("❌ 源拿不到：" + 源); continue; }
    if (UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(出) != null) { L("已存在，跳过：" + 出); continue; }

    var 拷 = UnityEngine.Object.Instantiate(原);          // 复制一份，避免改到 FBX 里的子资产
    拷.name = 表[i, 1];
    var 设置 = UnityEditor.AnimationUtility.GetAnimationClipSettings(拷);
    设置.loopTime = false;
    UnityEditor.AnimationUtility.SetAnimationClipSettings(拷, 设置);
    UnityEditor.AssetDatabase.CreateAsset(拷, 出);
    L("✔ " + 表[i, 1] + "：源 " + 原.length.ToString("F2") + "s / 曲线 " + UnityEditor.AnimationUtility.GetCurveBindings(原).Length
      + " → " + 出);
}
UnityEditor.AssetDatabase.SaveAssets();
UnityEditor.AssetDatabase.Refresh();

L("");
L("复查（组件的取法）：");
foreach (var n in new string[] { "八九玄功/长刀女_Attack1", "八九玄功/长刀女_Attack2", "八九玄功/杨戬_Attack1", "八九玄功/杨戬_Attack2",
                                 "八九玄功/长刀女_Attack1_御风", "八九玄功/长刀女_Attack2_御风", "八九玄功/杨戬_Attack1_御风", "八九玄功/杨戬_Attack2_御风" })
{
    var c = Resources.Load<AnimationClip>("技能动作/" + n);
    L("   " + n.PadRight(30) + " → " + (c == null ? "❌ null" : "✅ " + c.length.ToString("F2") + "s 曲线 " + UnityEditor.AnimationUtility.GetCurveBindings(c).Length + " humanMotion=" + c.humanMotion));
}
return sb.ToString();
