// 最小对照：同一个副本上，分别采样 (a) FBX 里的攻击片段 (b) 刚烘的片段，看手动不动
var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");

var 副本 = UnityEngine.Object.Instantiate(GameObject.Find("Player"));
副本.name = "采样对照";
副本.transform.position = new Vector3(0f, -600f, 0f);
副本.transform.rotation = Quaternion.identity;
foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
var an = 副本.GetComponentInChildren<Animator>(true);
an.runtimeAnimatorController = null;
L("Animator 所在对象 = " + an.gameObject.name + "（父 " + an.transform.parent.name + "）avatar=" + (an.avatar == null ? "null" : an.avatar.name) + " enabled=" + an.enabled);
Transform 手 = null;
foreach (var t in 副本.GetComponentsInChildren<Transform>(true)) if (t.name == "R_Hand") 手 = t;

AnimationClip 取FBX(string 路径)
{
    foreach (var o in UnityEditor.AssetDatabase.LoadAllAssetsAtPath(路径))
    {
        var c = o as AnimationClip;
        if (c != null && !c.name.StartsWith("__preview__")) return c;
    }
    return null;
}

var 甲 = 取FBX("Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack1.FBX");
L("FBX 片段 = " + (甲 == null ? "❌" : 甲.name + " len=" + 甲.length.ToString("F2") + " humanMotion=" + 甲.humanMotion + " 曲线=" + UnityEditor.AnimationUtility.GetCurveBindings(甲).Length));
L("FBX 片段采样：");
foreach (var 比例 in new float[] { 0f, 0.3f, 0.6f, 0.9f })
{
    甲.SampleAnimation(an.gameObject, 甲.length * 比例);
    L("   " + (比例 * 100f).ToString("F0") + "%：手=" + 手.position.ToString("F4") + " 手本地旋转=" + 手.localRotation.eulerAngles.ToString("F1"));
}

var 乙 = Resources.Load<AnimationClip>("技能动作/八九玄功/长刀女_Attack1");
L("烘的片段 = " + (乙 == null ? "❌" : 乙.name + " len=" + 乙.length.ToString("F2") + " humanMotion=" + 乙.humanMotion + " 曲线=" + UnityEditor.AnimationUtility.GetCurveBindings(乙).Length));
L("烘的片段采样（应该仍不动，验证问题在片段本身）：");
foreach (var 比例 in new float[] { 0f, 0.3f, 0.6f, 0.9f })
{
    乙.SampleAnimation(an.gameObject, 乙.length * 比例);
    L("   " + (比例 * 100f).ToString("F0") + "%：手=" + 手.position.ToString("F4"));
}
UnityEngine.Object.DestroyImmediate(副本);
return sb.ToString();
