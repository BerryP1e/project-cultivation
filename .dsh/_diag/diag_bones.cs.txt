// 全量对比：采样"新片段"与"御风 Idle"后，哪些骨骼的本地旋转不同？
// 这决定了"下半身取御风"能不能只靠挑肌肉做到
const string 御风Idle路径 = "Assets/Animations/凭虚御风/御风_Idle.anim";
const string 新路径 = "Assets/Animations/八九玄功/长刀女_Attack1_御风.anim";
const string 攻路径 = "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack1.FBX";
var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");

var 副本 = UnityEngine.Object.Instantiate(GameObject.Find("Player"));
副本.transform.position = new Vector3(0f, -500f, 0f);
副本.transform.rotation = Quaternion.identity;
foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
var an = 副本.GetComponentInChildren<Animator>(true);
an.runtimeAnimatorController = null;
var 骨骼 = 副本.GetComponentsInChildren<Transform>(true);
string 路径(Transform t)
{
    string p = t.name; var x = t.parent;
    while (x != null) { p = x.name + "/" + p; x = x.parent; }
    return p;
}
var 新 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(新路径);
var 御 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(御风Idle路径);
var 攻 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(攻路径);
float t = 0.833f;

System.Action<AnimationClip, float> 采样 = (c, tt) => c.SampleAnimation(an.gameObject, tt);
System.Collections.Generic.Dictionary<string, Quaternion> 快照()
{
    var d = new System.Collections.Generic.Dictionary<string, Quaternion>();
    foreach (var b in 骨骼) d[路径(b)] = b.localRotation;
    return d;
}

采样(新, t); var 快新 = 快照();
采样(御, Mathf.Repeat(t, 御.length)); var 快御 = 快照();
采样(攻, t); var 快攻 = 快照();

L("=== 新片段 vs 御风 Idle：本地旋转差 > 0.5° 的骨骼 ===");
foreach (var k in 快新.Keys)
{
    if (!快御.ContainsKey(k)) continue;
    float d = Quaternion.Angle(快新[k], 快御[k]);
    if (d > 0.5f) L("   " + k + " 差 " + d.ToString("F2") + "°");
}
L("");
L("=== 纯攻击 vs 御风 Idle：本地旋转差 > 0.5° 的骨骼（看上半身骨架命名）===");
int 打了 = 0;
foreach (var k in 快攻.Keys)
{
    if (!快御.ContainsKey(k)) continue;
    float d = Quaternion.Angle(快攻[k], 快御[k]);
    if (d > 0.5f) { L("   " + k + " 差 " + d.ToString("F2") + "°"); 打了++; if (打了 > 25) { L("   …"); break; } }
}
UnityEngine.Object.DestroyImmediate(副本);
return sb.ToString();
