// 烘焙四个「上半身=攻击、下半身=御风 Idle」的人形片段
// 做法（都用 muscle 曲线，人形片段可跨骨架自动重定向）：
//   · 下半身肌肉（Upper Leg / Lower Leg / Foot / Toes）+ RootT/RootQ ← 取 御风_Idle
//   · 其余肌肉（Spine / Chest / Neck / Head / 肩臂手 / 手指 / 眼 / 颌） ← 取攻击片段
// 御风 Idle 按时长循环采样（攻击通常比 idle 长一点）
const string 御风Idle路径 = "Assets/Animations/凭虚御风/御风_Idle.anim";
const string 输出目录 = "Assets/Animations/八九玄功";
const int Fps = 30;

string[] 源路径 = new string[] {
    "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack1.FBX",
    "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack2.FBX",
    "Assets/resources/NPC/Human/YangJian_Animation/YangJian@Attack1.FBX",
    "Assets/resources/NPC/Human/YangJian_Animation/YangJian@Attack2.FBX" };
string[] 名表 = new string[] { "长刀女_Attack1", "长刀女_Attack2", "杨戬_Attack1", "杨戬_Attack2" };

var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");

System.Func<string, bool> 是下半身 = (n) =>
    n.Contains("Upper Leg") || n.Contains("Lower Leg") || n.Contains("Foot") || n.Contains("Toes");
System.Func<string, bool> 是根 = (n) => n.StartsWith("RootT") || n.StartsWith("RootQ");

System.Collections.Generic.Dictionary<string, AnimationCurve> 读曲线(AnimationClip c)
{
    var d = new System.Collections.Generic.Dictionary<string, AnimationCurve>();
    foreach (var b in UnityEditor.AnimationUtility.GetCurveBindings(c))
    {
        if (b.path != "") continue;                       // 人形片段：肌肉曲线 path 为空
        var cur = UnityEditor.AnimationUtility.GetEditorCurve(c, b);
        if (cur != null) d[b.propertyName] = cur;
    }
    return d;
}

if (!UnityEditor.AssetDatabase.IsValidFolder(输出目录))
    UnityEditor.AssetDatabase.CreateFolder("Assets/Animations", "八九玄功");

var 御风 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(御风Idle路径);
if (御风 == null) { L("找不到 " + 御风Idle路径); return sb.ToString(); }
var 御风曲线 = 读曲线(御风);
L("御风 Idle：length=" + 御风.length.ToString("F2") + "s humanMotion=" + 御风.humanMotion + " 曲线=" + 御风曲线.Count);
L("下半身肌肉名（从 Unity 的标准列表里挑出来的）：");
int 下半身数 = 0;
foreach (var mn in HumanTrait.MuscleName) if (是下半身(mn)) { 下半身数++; L("   " + mn); }
L("下半身肌肉共 " + 下半身数 + " 条 / 全部肌肉 " + HumanTrait.MuscleCount + " 条");
L("");

for (int i = 0; i < 源路径.Length; i++)
{
    var 源 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(源路径[i]);
    if (源 == null) { L("拿不到 " + 源路径[i]); continue; }
    var 源曲线 = 读曲线(源);

    int 帧数 = Mathf.Max(2, Mathf.RoundToInt(源.length * Fps));
    var 产出 = new AnimationClip { name = 名表[i] + "_御风", frameRate = Fps };

    int 写了 = 0, 来自御风 = 0;
    for (int m = 0; m < HumanTrait.MuscleCount; m++)
    {
        string mn = HumanTrait.MuscleName[m];
        bool 用御风 = 是下半身(mn);
        var 首选 = 用御风 ? 御风曲线 : 源曲线;
        var 备用 = 用御风 ? 源曲线 : 御风曲线;
        AnimationCurve 曲线源; bool 有 = 首选.TryGetValue(mn, out 曲线源);
        if (!有) 有 = 备用.TryGetValue(mn, out 曲线源);
        if (!有) continue;                                   // 两边都没有 → 不写（= 中性值）
        if (用御风) 来自御风++;

        var 曲线 = new AnimationCurve();
        for (int f = 0; f <= 帧数; f++)
        {
            float t = 源.length * f / 帧数;
            float 采样t = 用御风 && 御风.length > 0.01f ? Mathf.Repeat(t, 御风.length) : t;
            float v = 首选.ContainsKey(mn) ? 首选[mn].Evaluate(采样t) : 曲线源.Evaluate(采样t);
            曲线.AddKey(new Keyframe(t, v));
        }
        UnityEditor.AnimationUtility.SetEditorCurve(产出,
            UnityEditor.EditorCurveBinding.FloatCurve("", typeof(Animator), mn), 曲线);
        写了++;
    }

    // 根：御风优先
    foreach (var prop in new string[] { "RootT.x", "RootT.y", "RootT.z", "RootQ.x", "RootQ.y", "RootQ.z", "RootQ.w" })
    {
        AnimationCurve c0;
        bool 有御风 = 御风曲线.TryGetValue(prop, out c0);
        if (!有御风 && !源曲线.TryGetValue(prop, out c0)) continue;
        var 曲线 = new AnimationCurve();
        for (int f = 0; f <= 帧数; f++)
        {
            float t = 源.length * f / 帧数;
            float 采样t = 有御风 && 御风.length > 0.01f ? Mathf.Repeat(t, 御风.length) : t;
            float v = 有御风 ? c0.Evaluate(采样t) : (源曲线.ContainsKey(prop) ? 源曲线[prop].Evaluate(t) : 0f);
            曲线.AddKey(new Keyframe(t, v));
        }
        UnityEditor.AnimationUtility.SetEditorCurve(产出,
            UnityEditor.EditorCurveBinding.FloatCurve("", typeof(Animator), prop), 曲线);
        写了++;
    }

    var 设置 = UnityEditor.AnimationUtility.GetAnimationClipSettings(产出);
    设置.loopTime = false;
    UnityEditor.AnimationUtility.SetAnimationClipSettings(产出, 设置);

    string 出路径 = 输出目录 + "/" + 名表[i] + "_御风.anim";
    var 旧 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(出路径);
    if (旧 != null) UnityEditor.AssetDatabase.DeleteAsset(出路径);
    UnityEditor.AssetDatabase.CreateAsset(产出, 出路径);
    L("✔ " + 名表[i] + "_御风.anim：" + 源.length.ToString("F2") + "s / " + 帧数 + " 帧 / 曲线 " + 写了
      + " 条（下半身来自御风 " + 来自御风 + " 条）");
}
UnityEditor.AssetDatabase.SaveAssets();
UnityEditor.AssetDatabase.Refresh();

// ---- 抽查：新片段的下半身是否等于御风、上半身是否等于攻击 ----
L("");
L("=== 抽查（曲线逐点比）===");
var 新1 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(输出目录 + "/" + 名表[0] + "_御风.anim");
var 攻1 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(源路径[0]);
if (新1 != null && 攻1 != null)
{
    var 新曲 = 读曲线(新1);
    var 攻曲 = 读曲线(攻1);
    float t = 新1.length * 0.5f;
    foreach (var mn in new string[] { "Left Upper Leg Front-Back", "Right Lower Leg Stretch", "Left Foot Up-Down" })
    {
        float a = 新曲.ContainsKey(mn) ? 新曲[mn].Evaluate(t) : 0f;
        float b = 御风曲线.ContainsKey(mn) ? 御风曲线[mn].Evaluate(Mathf.Repeat(t, 御风.length)) : 0f;
        L("  下半身 " + mn + "：新=" + a.ToString("F4") + " 御风=" + b.ToString("F4") + (Mathf.Abs(a - b) < 1e-4f ? " ✅" : " ❌"));
    }
    foreach (var mn in new string[] { "Right Arm Front-Back", "Spine Front-Back", "Head Nod Down-Up" })
    {
        float a = 新曲.ContainsKey(mn) ? 新曲[mn].Evaluate(t) : 0f;
        float b = 攻曲.ContainsKey(mn) ? 攻曲[mn].Evaluate(t) : 0f;
        L("  上半身 " + mn + "：新=" + a.ToString("F4") + " 攻击=" + b.ToString("F4") + (Mathf.Abs(a - b) < 1e-4f ? " ✅" : " ❌"));
    }
}
return sb.ToString();
