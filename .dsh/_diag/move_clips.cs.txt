// 把四个「御风适配」片段从 Assets/Animations/八九玄功/ 挪到 Assets/resources/技能动作/八九玄功/
// 理由：普攻组件走 Resources.Load("技能动作/xxx")（现有普攻_远程_01 就是这条约定）
var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");
const string 源目录 = "Assets/Animations/八九玄功";
const string 目标父 = "Assets/resources/技能动作";
const string 目标目录 = "Assets/resources/技能动作/八九玄功";

if (!UnityEditor.AssetDatabase.IsValidFolder(目标目录))
{
    var 结果 = UnityEditor.AssetDatabase.CreateFolder(目标父, "八九玄功");
    L("建目录 " + 目标目录 + " → " + (string.IsNullOrEmpty(结果) ? "失败" : 结果));
}

string[] 名 = new string[] { "长刀女_Attack1_御风.anim", "长刀女_Attack2_御风.anim", "杨戬_Attack1_御风.anim", "杨戬_Attack2_御风.anim" };
foreach (var n in 名)
{
    string 从 = 源目录 + "/" + n;
    string 到 = 目标目录 + "/" + n;
    if (UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(到) != null) { L("已存在，跳过：" + 到); continue; }
    if (UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(从) == null) { L("源头没有：" + 从); continue; }
    var err = UnityEditor.AssetDatabase.MoveAsset(从, 到);
    L((string.IsNullOrEmpty(err) ? "✔ 移动成功 " : "❌ 移动失败(" + err + ") ") + 从 + " → " + 到);
}
UnityEditor.AssetDatabase.SaveAssets();
UnityEditor.AssetDatabase.Refresh();

// 挪完复查：Resources.Load 真能取到吗（这才是将来组件要走的路）
L("");
L("复查（模拟组件取片段的方式）：");
foreach (var n in new string[] { "八九玄功/长刀女_Attack1_御风", "八九玄功/长刀女_Attack2_御风", "八九玄功/杨戬_Attack1_御风", "八九玄功/杨戬_Attack2_御风" })
{
    var c = Resources.Load<AnimationClip>("技能动作/" + n);
    L("   Resources.Load<AnimationClip>(\"技能动作/" + n + "\") → " + (c == null ? "❌ null" : "✅ " + c.name + " " + c.length.ToString("F2") + "s 曲线 " + UnityEditor.AnimationUtility.GetCurveBindings(c).Length));
}
L("");
L("源目录还剩什么：");
foreach (var g in UnityEditor.AssetDatabase.FindAssets("t:AnimationClip", new string[] { 源目录 }))
    L("   " + UnityEditor.AssetDatabase.GUIDToAssetPath(g));
return sb.ToString();
