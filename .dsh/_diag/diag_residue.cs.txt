// 决定性测试：SampleAnimation 是不是"有残留"（同一片段连采两次结果是否一致；换片段后是否互相污染）
const string 御风Idle路径 = "Assets/Animations/凭虚御风/御风_Idle.anim";
const string 新路径 = "Assets/Animations/八九玄功/长刀女_Attack1_御风.anim";
var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");

var 副本 = UnityEngine.Object.Instantiate(GameObject.Find("Player"));
副本.transform.position = new Vector3(0f, -500f, 0f);
foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
var an = 副本.GetComponentInChildren<Animator>(true);
an.runtimeAnimatorController = null;
Transform 髋 = null;
foreach (var t in 副本.GetComponentsInChildren<Transform>(true)) if (t.name == "Hip") 髋 = t;
var 新 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(新路径);
var 御 = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(御风Idle路径);
float t = 0.833f;

System.Action<string> 打 = (标签) => L(标签 + "：Hip 本地旋转=" + 髋.localRotation.eulerAngles.ToString("F2")
    + " quat=" + 髋.localRotation.ToString("F4"));

新.SampleAnimation(an.gameObject, t); 打("新 #1");
新.SampleAnimation(an.gameObject, t); 打("新 #2");
御.SampleAnimation(an.gameObject, Mathf.Repeat(t, 御.length)); 打("御风（紧接新之后）");
御.SampleAnimation(an.gameObject, Mathf.Repeat(t, 御.length)); 打("御风 #2");
新.SampleAnimation(an.gameObject, t); 打("新（紧接御风之后）");
御.SampleAnimation(an.gameObject, Mathf.Repeat(t, 御.length)); 打("御风 #3");

// 换一个"只有上半身不同"的对照：把御风片段按 30% 采样（与 t=0.833 的 34.7% 接近，下半身差不多）
御.SampleAnimation(an.gameObject, 0.833f); 打("御风（直接 t=0.833）");
UnityEngine.Object.DestroyImmediate(副本);
return sb.ToString();
