// 量挥砍弧：刀身线段两端点换算到【玩家本地空间】，看它到底往哪边挥、够多远
// 同时打印角色朝向 / Player_Visual 的朝向偏移 / 到目标的方向
System.Func<string, Type> 找类型 = (名) =>
{
    foreach (var a in System.AppDomain.CurrentDomain.GetAssemblies())
    {
        var t = a.GetType(名);
        if (t != null) return t;
        foreach (var t2 in a.GetTypes()) if (t2.Name == 名) return t2;
    }
    return null;
};
var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");

System.Collections.IEnumerator 主()
{
    var 玩家 = GameObject.Find("Player");
    var pd = GameObject.Find("CharacterUI").GetComponent<UIPanelData>();
    var 功法 = UnityEditor.AssetDatabase.LoadAssetAtPath<GongFaDefinition>("Assets/Data/Generated/GongFaDefinition/gongfa_jiuba_xuangong.asset");
    pd.学会(功法); pd.当前功法 = 功法; pd.RaiseChanged();
    yield return new WaitForSecondsRealtime(0.8f);

    var 靶子 = UnityEngine.Object.Instantiate(Resources.Load<GameObject>("摆设/练功木桩"),
        玩家.transform.position + 玩家.transform.forward * 2.0f, Quaternion.identity);
    var 靶Npc = 靶子.GetComponentInChildren<NpcInstance>();
    玩家.GetComponent<NpcTargeting>().Lock(靶Npc);
    yield return new WaitForSecondsRealtime(1.4f);

    Transform 视觉 = null, 武器根 = null;
    foreach (var t in 玩家.GetComponentsInChildren<Transform>(true))
    {
        if (t.name == "Player_Visual") 视觉 = t;
        if (t.name == "八九玄功_三尖两刃刀") 武器根 = t;
    }
    L("玩家根：pos=" + 玩家.transform.position.ToString("F2") + " forward=" + 玩家.transform.forward.ToString("F2") + " 欧拉=" + 玩家.transform.eulerAngles.ToString("F1"));
    if (视觉 != null) L("Player_Visual：本地欧拉=" + 视觉.localRotation.eulerAngles.ToString("F1") + " 世界forward=" + 视觉.forward.ToString("F2"));
    var 到靶 = (靶Npc.transform.position - 玩家.transform.position); 到靶.y = 0f;
    L("到靶子：世界方向=" + 到靶.normalized.ToString("F2") + " 距离=" + 到靶.magnitude.ToString("F2")
      + " ｜ 与玩家 forward 的夹角=" + Vector3.Angle(玩家.transform.forward, 到靶).ToString("F1") + "°");
    if (视觉 != null) L("          与 Player_Visual forward 的夹角=" + Vector3.Angle(视觉.forward, 到靶).ToString("F1") + "°");

    var smr = 武器根.GetComponentInChildren<SkinnedMeshRenderer>();
    var t普攻 = 找类型("BasicJiuba01");
    var 普攻 = 玩家.GetComponent(t普攻);
    var p已标定 = t普攻.GetProperty("当前是御风版");

    // 手动标定一次（组件内部会做，这里直接算本地端点）
    var 烘 = new Mesh(); smr.BakeMesh(烘);
    var 界 = 烘.bounds; var 半 = 界.size * 0.5f;
    int 轴 = (半.x >= 半.y && 半.x >= 半.z) ? 0 : (半.y >= 半.z ? 1 : 2);
    var d = 轴 == 0 ? new Vector3(半.x, 0, 0) : (轴 == 1 ? new Vector3(0, 半.y, 0) : new Vector3(0, 0, 半.z));
    var 本地A = 界.center - d; var 本地B = 界.center + d;
    UnityEngine.Object.DestroyImmediate(烘);
    L("刀身本地线段（渲染体空间）：" + 本地A.ToString("F2") + " → " + 本地B.ToString("F2"));

    t普攻.GetMethod("出手").Invoke(普攻, null);
    Vector3 Lmin = new Vector3(999, 999, 999), Lmax = new Vector3(-999, -999, -999);
    float 最小 = 999f;
    var 靶点 = 靶Npc.transform.position + Vector3.up * 0.9f;
    for (int i = 0; i < 20; i++)
    {
        yield return new WaitForSecondsRealtime(0.08f);
        var a = smr.transform.TransformPoint(本地A);
        var b = smr.transform.TransformPoint(本地B);
        foreach (var w in new Vector3[] { a, b })
        {
            var lp = 玩家.transform.InverseTransformPoint(w);
            Lmin = Vector3.Min(Lmin, lp); Lmax = Vector3.Max(Lmax, lp);
        }
        // 刀身线段到靶点的最近距离（粗略用两端点+中点）
        var mid = (a + b) * 0.5f;
        float dm = Mathf.Min(Vector3.Distance(a, 靶点), Mathf.Min(Vector3.Distance(b, 靶点), Vector3.Distance(mid, 靶点)));
        最小 = Mathf.Min(最小, dm);
    }
    L("");
    L("挥砍中刀身两端在【玩家本地空间】的范围：");
    L("   X(本地左右) " + Lmin.x.ToString("F2") + " ~ " + Lmax.x.ToString("F2"));
    L("   Y(本地上下) " + Lmin.y.ToString("F2") + " ~ " + Lmax.y.ToString("F2"));
    L("   Z(本地前后，+Z=角色 forward) " + Lmin.z.ToString("F2") + " ~ " + Lmax.z.ToString("F2"));
    L("   两端点到靶点（含中点）的最小距离 ≈ " + 最小.ToString("F2") + " 米");
    UnityEngine.Object.DestroyImmediate(靶子);
    Debug.Log("【挥砍弧诊断】\n" + sb.ToString());
}
return 主();
