// 最终验证（第三版）：武器缩放按"武器长÷身高 = 1.660"定，机位改为全身正面/侧面
const string 武器Prefab路径 = "Assets/resources/Weapons/八九玄功_三尖两刃刀.prefab";
const float 缩放 = 0.4911f;              // = 应有长 4.347 ÷ (4.9154 × 1.8007)
string[] 片段路径 = new string[] {
    "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack1.FBX",
    "Assets/resources/NPC/Human/ChangDaoNv_Animation/ChangDaoNv@Attack2.FBX",
    "Assets/resources/NPC/Human/YangJian_Animation/YangJian@Attack1.FBX",
    "Assets/resources/NPC/Human/YangJian_Animation/YangJian@Attack2.FBX" };

var sb = new System.Text.StringBuilder();
System.Action<string> L = (s) => sb.Append(s + "\n");
Transform 找(GameObject 根, string 名)
{
    foreach (var t in 根.GetComponentsInChildren<Transform>(true)) if (t.name == 名) return t;
    return null;
}
string 截图目录 = System.IO.Path.Combine(System.IO.Path.GetDirectoryName(Application.dataPath), "screenshots");

System.Action<GameObject, Vector3, string> 拍 = (目标, 偏移, 文件名) =>
{
    var camGo = new GameObject("临时相机");
    var cam = camGo.AddComponent<Camera>();
    cam.fieldOfView = 42f; cam.nearClipPlane = 0.05f; cam.farClipPlane = 200f;
    cam.clearFlags = CameraClearFlags.SolidColor;
    cam.backgroundColor = new Color(0.13f, 0.14f, 0.16f, 1f);
    camGo.transform.position = 目标.transform.position + 偏移;
    camGo.transform.LookAt(目标.transform.position + new Vector3(0f, 1.35f, 0f));
    var 灯Go = new GameObject("临时灯");
    var 灯 = 灯Go.AddComponent<Light>();
    灯.type = LightType.Directional; 灯.intensity = 1.45f; 灯.color = Color.white;
    灯Go.transform.rotation = Quaternion.Euler(45f, 40f, 0f);
    var rt = new RenderTexture(700, 950, 24);
    cam.targetTexture = rt; cam.Render();
    RenderTexture.active = rt;
    var tex = new Texture2D(700, 950, TextureFormat.RGB24, false);
    tex.ReadPixels(new Rect(0f, 0f, 700f, 950f), 0, 0); tex.Apply();
    RenderTexture.active = null; cam.targetTexture = null;
    System.IO.File.WriteAllBytes(System.IO.Path.Combine(截图目录, 文件名), tex.EncodeToPNG());
    UnityEngine.Object.DestroyImmediate(rt); UnityEngine.Object.DestroyImmediate(tex);
    UnityEngine.Object.DestroyImmediate(camGo); UnityEngine.Object.DestroyImmediate(灯Go);
};

var 真玩家 = GameObject.Find("Player");
var 副本 = UnityEngine.Object.Instantiate(真玩家);
副本.name = "临时验证_玩家";
副本.transform.position = new Vector3(0f, -500f, 0f);
副本.transform.rotation = Quaternion.identity;
foreach (var c in 副本.GetComponentsInChildren<MonoBehaviour>(true)) c.enabled = false;
var an = 副本.GetComponentInChildren<Animator>(true);
an.runtimeAnimatorController = null;
var 手 = 找(副本, "R_Hand");
var 武器 = UnityEngine.Object.Instantiate(UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>(武器Prefab路径));
武器.transform.SetParent(手, false);
武器.transform.localPosition = Vector3.zero;
武器.transform.localRotation = Quaternion.identity;
武器.transform.localScale = Vector3.one * 缩放;
var 武器Smr = 武器.GetComponentInChildren<SkinnedMeshRenderer>();
L("武器：缩放=" + 缩放.ToString("F4") + " 本地长轴=" + 武器Smr.localBounds.size.x.ToString("F3")
  + " 渲染体 lossyScale=" + 武器Smr.transform.lossyScale.ToString("F4")
  + " → 世界长 ≈ " + (武器Smr.localBounds.size.x * 武器Smr.transform.lossyScale.x).ToString("F3") + " 米"
  + "（杨戬本体 4.424 米）");
L("玩家身体网格高度=2.618 米 ⇒ 比例=" + (武器Smr.localBounds.size.x * 武器Smr.transform.lossyScale.x / 2.618f).ToString("F3") + "（杨戬 1.660）");

foreach (var p in 片段路径)
{
    var clip = UnityEditor.AssetDatabase.LoadAssetAtPath<AnimationClip>(p);
    if (clip == null) continue;
    string 短名 = System.IO.Path.GetFileNameWithoutExtension(p).Replace("@", "_");
    clip.SampleAnimation(an.gameObject, clip.length * 0.55f);
    L("");
    L("--- " + clip.name + " @55%：手=" + 手.position.ToString("F3"));
    拍(副本, new Vector3(-4.6f, 2.3f, -5.6f), "八九_" + 短名 + "_前右.png");
    拍(副本, new Vector3(-7.4f, 1.5f, 0f), "八九_" + 短名 + "_侧.png");
}
UnityEngine.Object.DestroyImmediate(武器);
UnityEngine.Object.DestroyImmediate(副本);
L("");
L("已出图（每个动作 2 张，前右 + 正侧），临时对象已销毁");
return sb.ToString();
