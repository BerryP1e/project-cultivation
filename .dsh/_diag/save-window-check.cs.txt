using UnityEngine.UI;
System.Collections.IEnumerator Run(){
var menu=Object.FindObjectOfType<MainMenuUI>();
if(menu==null)throw new System.Exception("请在开始场景的 Play 中运行");
var wasOpen=menu.存档面板.activeSelf;
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var oldMode=(bool)menu.GetType().GetField("新游戏模式",flags).GetValue(menu);
var output=new System.Text.StringBuilder();
void Check(bool ok,string message){if(!ok)throw new System.Exception(message);output.AppendLine(message+" OK");}
try{
foreach(bool newGame in new[]{false,true}){
menu.OpenSlots(newGame);yield return new WaitForSecondsRealtime(.8f);
foreach(var skin in Object.FindObjectsOfType<InkUIRuntimeSkin>())skin.Refresh();
Canvas.ForceUpdateCanvases();
var rows=menu.槽位容器.GetComponentsInChildren<Button>().Where(b=>b.name.StartsWith("Slot")).ToArray();
Check(rows.Length==SaveSystem.槽位数,"五个存档槽位完整");
var data=SaveSystem.读全部();
for(int i=0;i<rows.Length;i++){
var row=rows[i];bool occupied=data[i]!=null&&!data[i].是空的;
Check(row.interactable==(newGame?!occupied:occupied),"槽位"+(i+1)+"模式准入保留");
Check(row.GetComponent<Image>().sprite==null&&row.transform.Find("FineFrame")!=null,"槽位"+(i+1)+"规整细框、不再套墨迹标签");
var title=row.transform.Find("Name").GetComponent<Text>();
Check(title.text==(occupied?data[i].角色名+"    "+data[i].境界:"暂无存档"),"槽位"+(i+1)+"身份和境界完整");
if(occupied){
Check(row.transform.Find("Time").GetComponent<Text>().text==data[i].最后存档时间,"原存档时间完整");
Check(row.transform.Find("Delete"+i).GetComponent<Button>().onClick.GetPersistentEventCount()==0,"删除使用原运行时监听");
}
Vector3[] corners=new Vector3[4];row.GetComponent<RectTransform>().GetWorldCorners(corners);
Check(corners.All(v=>v.x>=0&&v.x<=Screen.width&&v.y>=0&&v.y<=Screen.height),"槽位"+(i+1)+"位于屏幕内");
}
Check(menu.存档面板.GetComponent<Image>().sprite==null,"窗口为纯色底和细回纹框");
ScreenCapture.CaptureScreenshot("screenshots/save-window-"+(newGame?"new":"load")+"-"+Screen.width+"x"+Screen.height+".png");yield return new WaitForSecondsRealtime(.4f);
}
// 只点第一次删除来验证确认状态，绝不触发实际删除。
menu.OpenSlots(false);var existing=SaveSystem.读全部();int index=System.Array.FindIndex(existing,d=>d!=null&&!d.是空的);
if(index>=0){
menu.槽位容器.Find("Slot"+index+"/Delete"+index).GetComponent<Button>().onClick.Invoke();
Check(menu.槽位容器.Find("Slot"+index+"/Delete"+index+"/Label").GetComponent<Text>().text=="确认删除","第一次删除只进入确认");
Check(SaveSystem.读档(index).最后存档时间==existing[index].最后存档时间,"验证未改写或删除存档");
}
Debug.Log(output.ToString());
System.IO.File.WriteAllText("screenshots/save-window-check-"+Screen.width+"x"+Screen.height+".txt",output.ToString());
}finally{menu.OpenSlots(oldMode);if(!wasOpen)menu.CloseSlots();}
}
return Run();
