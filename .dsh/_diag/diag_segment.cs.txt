System.Func<string, Type> 找类型 = (名) =>
{
    foreach (var a in System.AppDomain.CurrentDomain.GetAssemblies())
    {
        var t = a.GetType(名);
        if (t != null) return t;
        foreach (var t2 in a.GetTypes()) if (t2.Name == 名) return t2;
    }
    return null;
};
var sb = new System.Text.StringBuilder();

Quaternion QuaternionFrom(Matrix4x4 m) => m.rotation;
float 线段距离(Vector3 p1, Vector3 p2, Vector3 q1, Vector3 q2)
{
    var d1 = p2 - p1; var d2 = q2 - q1; var r = p1 - q1;
    float a = Vector3.Dot(d1, d1), e = Vector3.Dot(d2, d2), f = Vector3.Dot(d2, r), s, t;
    const float 极小 = 1e-8f;
    if (a <= 极小 && e <= 极小) { s = t = 0f; }
    else if (a <= 极小) { s = 0f; t = Mathf.Clamp01(f / e); }
    else { float c = Vector3.Dot(d1, r);
        if (e <= 极小) { t = 0f; s = Mathf.Clamp01(-c / a); }
        else { float b = Vector3.Dot(d1, d2); float 分母 = a * e - b * b;
            s = 分母 != 0f ? Mathf.Clamp01((b * f - c * e) / 分母) : 0f; t = (b * s + f) / e;
            if (t < 0f) { t = 0f; s = Mathf.Clamp01(-c / a); } else if (t > 1f) { t = 1f; s = Mathf.Clamp01((b - c) / a); } } }
    return Vector3.Distance(p1 + d1 * s, q1 + d2 * t);
}
