using UnityEngine.UI;
System.Collections.IEnumerator Run(){
var menu=Object.FindObjectOfType<PlayerStatsDebugPanel>();bool visible=menu.Visible;
var marks=对话标记.全部标记();var data=Object.FindObjectOfType<UIPanelData>();var items=data.物品.ToArray();

var sb=new System.Text.StringBuilder();void Check(bool ok,string text){if(!ok)throw new System.Exception(text);sb.AppendLine(text+" OK");}
try{
menu.Visible=false;yield return new WaitForSecondsRealtime(.4f);
var mask=view.GetComponentsInChildren<RectTransform>(true).First(t=>t.name=="RecipeUnrollMask");
var tube=view.GetComponentsInChildren<Image>(true).First(t=>t.name=="OpeningRecipeTube");

Check(view.选中丹方==all[1].id && mask.gameObject.activeSelf && mask.rect.width<815,"新丹方再展开");

Check(lid.sizeDelta==new Vector2(505,168) && lid.anchorMin==new Vector2(.5f,.86f),"盖面按有效轮廓缩小并上移贴合炉口");

ScreenCapture.CaptureScreenshot("screenshots/alchemy-debug-"+Screen.width+"x"+Screen.height+".png");yield return null;

view.关闭();Object.Destroy(view.gameObject);menu.Visible=visible;对话标记.清空();foreach(var mark in marks)对话标记.添加(mark);if(quest!=null)quest.enabled=questEnabled;
}
}
