using System;
using System.IO;
using System.Linq;
using System.Collections;
using UnityEditor;
using UnityEngine;
using UnityEngine.SceneManagement;
using Cultivation.WorldAuthoring;
IEnumerator Run() {
 if(string.IsNullOrEmpty(SessionState.GetString(SaveSystem.验证目录键,"")))throw new Exception("Requires isolated save directory");
 var report=new System.Collections.Generic.List<string>();
 void Check(bool ok,string text){if(!ok)throw new Exception(text);report.Add(text);}
 IEnumerator Frames(int count){for(int i=0;i<count;i++){EditorApplication.QueuePlayerLoopUpdate();yield return null;}}
 IEnumerator WaitScene(string name){float timeout=Time.realtimeSinceStartup+25;while((SceneManager.GetActiveScene().name!=name||黑幕字幕.过渡进行中)&&Time.realtimeSinceStartup<timeout){EditorApplication.QueuePlayerLoopUpdate();yield return null;}Check(SceneManager.GetActiveScene().name==name&&!黑幕字幕.过渡进行中,"transition completed: "+name);yield return Frames(12);}
 var fresh=SaveSystem.建新档("洞府回归");SaveSystem.当前存档=fresh;SaveSystem.当前槽位=4;SaveSystem.存档(4,fresh);SaveApplier.开始读取存档();
 var load=SceneManager.LoadSceneAsync("3C_Testbed");while(!load.isDone)yield return null;yield return Frames(15);
 var player=GameObject.Find("Player");var bag=UnityEngine.Object.FindObjectOfType<UIPanelData>();
 Check(player&&player.GetComponent<SaveApplier>(),"player owns independent save application");
 Check(!UnityEngine.Object.FindObjectsOfType<MonoBehaviour>().Any(c=>new[]{"任务管理器","主线开场","任务引导"}.Contains(c.GetType().Name)),"no story manager or forced opening in home");
 Check(bag.当前法宝&&bag.当前法宝.法宝id==TowerRealmTreasure.法宝id&&bag.已拥有(bag.当前法宝),"fresh character owns and equips tower treasure");
 Check(fresh.背包物品.Count(id=>id.StartsWith("item_home_")&&id.EndsWith("_hall"))==5,"five functional building tokens in new character inventory");
 foreach(var id in new[]{"alchemy_hall","merit_hall","technique_hall","deacon_hall","forge_hall"}){
  var definition=摆放物库.取("home_"+id);var token=QuestDatabase.取().找物品("item_home_"+id);
  Check(definition!=null&&token&&token.使用效果 is 摆放物件效果,"token uses existing placement transaction: "+id);
  var prefab=Resources.Load<GameObject>(definition.模型路径);Check(prefab&&prefab.GetComponent<StationInteractable>(),"reusable interactive home prefab: "+id);
  Check(prefab.GetComponentsInChildren<Renderer>(true).Any(r=>r.enabled),"building visuals enabled: "+id);
  if(id=="alchemy_hall")Check(prefab.GetComponent<StationInteractable>().界面预制体,"original manual alchemy UI retained");
  if(id=="merit_hall"||id=="technique_hall")Check(prefab.GetComponent<功德堂兑换>()&&prefab.GetComponent<功德堂兑换>().传法阁==(id=="technique_hall"),"exchange role preserved: "+id);
  if(id=="deacon_hall")Check(prefab.GetComponent<执事阁界面>(),"commission panel retained");
 }
 var pc=player.GetComponent<PlayerController>();pc.enabled=false;var cc=player.GetComponent<CharacterController>();cc.enabled=false;
 var origin=player.transform.position+new Vector3(1.25f,0,1.75f);player.transform.SetPositionAndRotation(origin,Quaternion.Euler(0,73,0));
 var tokenBefore=bag.物品.Count;
 Check(TowerRealmTreasure.使用(player),"tower treasure enters from home");yield return WaitScene(TowerRealmTreasure.塔场景);
 var tower=UnityEngine.Object.FindObjectOfType<TowerController>();Check(tower&&tower.塔运行中&&tower.本层存活数>0,"native tower monsters spawn on entry");
 Check(UnityEngine.Object.FindObjectOfType<UIPanelData>().物品.Count==tokenBefore,"inventory survives tower entry");
 foreach(var zone in tower.刷怪区)foreach(var npc in zone.GetComponentsInChildren<NpcInstance>())if(npc&&!npc.IsDead)npc.GetType();
 var current=tower.当前层;tower.下一层();yield return Frames(8);Check(tower.当前层==current+1&&tower.本层存活数>0,"floor advancement refreshes native enemies");
 var saved=new SaveData();SaveSystem.从角色采集(saved);Check(saved.镇妖阁返回点!=null&&Vector3.Distance(saved.镇妖阁返回点.位置,origin)<.01f,"tower save includes real-world origin");
 SaveSystem.存档(4,saved);Check(JsonUtility.FromJson<SaveData>(JsonUtility.ToJson(saved)).镇妖阁返回点!=null,"return point survives save JSON round trip");
 Check(TowerRealmTreasure.使用(GameObject.Find("Player")),"E action exits during active combat");yield return WaitScene("3C_Testbed");
 player=GameObject.Find("Player");Check(Vector2.Distance(new Vector2(player.transform.position.x,player.transform.position.z),new Vector2(origin.x,origin.z))<.15f,"normal exit returns to exact summon position");
 Check(UnityEngine.Object.FindObjectOfType<UIPanelData>().物品.Count==tokenBefore,"inventory survives return");
 Check(TowerRealmTreasure.使用(player),"second entry succeeds");yield return WaitScene(TowerRealmTreasure.塔场景);
 tower=UnityEngine.Object.FindObjectOfType<TowerController>();
 typeof(TowerController).GetMethod("处理玩家死亡",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).Invoke(tower,new object[]{UnityEngine.Object.FindObjectOfType<PlayerVitals>()});
 yield return WaitScene("3C_Testbed");Check(!UnityEngine.Object.FindObjectOfType<PlayerVitals>().IsDead,"death exit produces living home player");
 Check(Vector2.Distance(new Vector2(GameObject.Find("Player").transform.position.x,GameObject.Find("Player").transform.position.z),new Vector2(origin.x,origin.z))<.15f,"death returns to summon position");
 Directory.CreateDirectory("../.local/diagnostics/idle-refocus");File.WriteAllLines("../.local/diagnostics/idle-refocus/home-tower-report.txt",report);Debug.Log("HOME_TOWER_RUNTIME_PASS "+report.Count);
}
return Run();
