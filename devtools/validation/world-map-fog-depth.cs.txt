using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine.SceneManagement;
using System.IO;
using Cultivation.WorldAuthoring;
if(Application.isPlaying)throw new Exception("Requires stopped Editor");Directory.CreateDirectory("../.local/diagnostics/map-repair");
var scene=EditorSceneManager.NewPreviewScene();var cameraRoot=new GameObject("Off-centre fog test",typeof(Camera));SceneManager.MoveGameObjectToScene(cameraRoot,scene);var camera=cameraRoot.GetComponent<Camera>();camera.scene=scene;camera.enabled=false;camera.depthTextureMode=DepthTextureMode.Depth;camera.allowHDR=true;camera.allowMSAA=true;camera.transform.position=new Vector3(0,10,-20);camera.transform.LookAt(new Vector3(0,6,0));camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=new Color(.01f,.02f,.03f);camera.farClipPlane=500;var root=new GameObject("Fog world");SceneManager.MoveGameObjectToScene(root,scene);var world=root.AddComponent<WorldMapRenderer>();var d=WorldMapData.Empty();d.bottomFogOnly=true;d.fogBaseHeight=-.7f;var palette=AssetDatabase.LoadAssetAtPath<WorldMapPalette>(WorldMapEditorBuilder.PalettePath);world.Configure(palette,d);var effect=cameraRoot.AddComponent<WorldMapCamera>();effect.world=world;var rt=new RenderTexture(640,480,24,RenderTextureFormat.ARGBHalf);rt.Create();camera.targetTexture=rt;var previous=RenderTexture.active;var mats=new System.Collections.Generic.List<Material>();var points=new[]{new Vector3(-5,9,2),new Vector3(5,2,0),new Vector3(0,16,9)};var colours=new[]{Color.red,Color.green,Color.blue};
try{
for(int i=0;i<points.Length;i++){var cube=GameObject.CreatePrimitive(PrimitiveType.Cube);SceneManager.MoveGameObjectToScene(cube,scene);cube.transform.position=points[i];cube.transform.localScale=Vector3.one*3;var mat=new Material(Shader.Find("Cultivation/WorldAuthoring/Surface"));mat.SetColor("_BaseColor",colours[i]);mat.SetFloat("_GrassBlend",0);mat.SetFloat("_NoiseStrength",0);cube.GetComponent<Renderer>().sharedMaterial=mat;mats.Add(mat);}
typeof(WorldMapCamera).GetMethod("OnPreCull",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance).Invoke(effect,null);camera.Render();RenderTexture.active=rt;var image=new Texture2D(640,480,TextureFormat.RGB24,false);image.ReadPixels(new Rect(0,0,640,480),0,0);image.Apply();File.WriteAllBytes("../.local/diagnostics/map-repair/fog-asymmetric.png",image.EncodeToPNG());for(int i=0;i<points.Length;i++){var p=camera.WorldToViewportPoint(points[i]);if(p.y>1||p.y<0)continue;var sample=image.GetPixel(Mathf.RoundToInt(p.x*640),Mathf.RoundToInt(p.y*480));Debug.Log("off-centre "+i+" uv="+p+" colour="+sample);if((i==0&&(sample.g>.03f||sample.b>.03f))||(i==1&&(sample.r>.03f||sample.b>.03f))||(i==2&&(sample.r>.03f||sample.g>.03f)))throw new Exception("Fog covers above-cloud opaque geometry away from screen centre: "+i+" "+sample);}Object.DestroyImmediate(image);Debug.Log("ASYMMETRIC_FOG_PASS");
}finally{RenderTexture.active=previous;camera.targetTexture=null;rt.Release();Object.DestroyImmediate(rt);foreach(var mat in mats)Object.DestroyImmediate(mat);EditorSceneManager.ClosePreviewScene(scene);}
