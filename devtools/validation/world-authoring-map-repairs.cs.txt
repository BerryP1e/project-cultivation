  if(yaw==0){world.Configure(palette,d);var root=(GameObject)typeof(WorldMapRenderer).GetMethod("InstantiateEntry",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance).Invoke(world,new object[]{entry,placed,yaw,scale,true,1f,float.NaN});Physics.SyncTransforms();foreach(float end in new[]{-.43f,.43f}){var p=(alongX?new Vector3(size.x*end,0,0):new Vector3(0,0,size.y*end))+new Vector3(entry.FootprintOffset.x,0,entry.FootprintOffset.y)*scale;p.y=8;var hits=root.GetComponentsInChildren<Collider>().Select(c=>{bool found=c.Raycast(new Ray(p,Vector3.down),out var h,20);return new{found,hit=h};}).Where(x=>x.found&&x.hit.normal.y>.55f).ToArray();Check(hits.Length>0&&Mathf.Abs(hits.Max(x=>x.hit.point.y))<.55f,"actual bridge deck connects shore: "+entry.label+" end "+end+" hits="+string.Join(",",hits.Select(x=>x.hit.point.y))+" pos="+placed+" deck="+entry.bridgeDeckHeight);}Object.DestroyImmediate(root);}

}finally{Object.DestroyImmediate(host);EditorSceneManager.ClosePreviewScene(preview);}
